## Python Internals & Memory Model

### How Python Memory Works

```python
# Python uses reference counting + garbage collection

import sys

# Check object size
x = 42
print(sys.getsizeof(x))  # ~28 bytes

y = "hello"
print(sys.getsizeof(y))  # ~54 bytes

# Reference counting
a = [1, 2, 3]
b = a  # Same object, reference count = 2
c = a.copy()  # Different object, c's ref count = 1

del a  # Decrements reference count
# When ref count = 0, object is garbage collected

# Important implications:
# 1. Modifying mutable objects affects all references
list_a = [1, 2, 3]
list_b = list_a  # list_b and list_a point to same list
list_a.append(4)
print(list_b)  # [1, 2, 3, 4] ← list_b affected!

# Solution: Deep copy for independent copies
import copy
list_c = copy.deepcopy(list_a)
list_a.append(5)
print(list_c)  # [1, 2, 3, 4] ← unchanged
```

### Global Interpreter Lock (GIL)

```python
# GIL: Only one thread executes Python bytecode at a time
import threading
import time

def cpu_bound_work():
    """CPU-intensive task"""
    total = 0
    for i in range(100_000_000):
        total += i
    return total

# Single-threaded
start = time.time()
cpu_bound_work()
cpu_bound_work()
print(f"Single-threaded: {time.time() - start:.2f}s")  # ~10 seconds

# Multi-threaded (doesn't help for CPU work!)
start = time.time()
threads = [
    threading.Thread(target=cpu_bound_work),
    threading.Thread(target=cpu_bound_work)
]
for t in threads:
    t.start()
for t in threads:
    t.join()
print(f"Multi-threaded: {time.time() - start:.2f}s")  # ~10 seconds (not faster!)

# Why? GIL ensures only one thread runs Python at a time
# Multi-threading HELPS for I/O-bound work (waiting on network, disk)
# Multi-processing HELPS for CPU-bound work (each process has own GIL)

# Solution for CPU-bound: Use multiprocessing
from multiprocessing import Pool

with Pool(2) as pool:
    start = time.time()
    pool.map(cpu_bound_work, [None, None])
    print(f"Multi-process: {time.time() - start:.2f}s")  # ~5 seconds (2x faster)
```

### Generator vs. List

```python
# Generators: Lazy evaluation, memory efficient
def infinite_sequence():
    """Generate infinite sequence"""
    num = 0
    while True:
        yield num
        num += 1

# List: Eager evaluation, memory intensive
numbers_list = list(range(1_000_000))  # ~40 MB in memory
numbers_gen = (x for x in range(1_000_000))  # ~0 bytes until consumed

# Generator example in data processing
def read_large_csv(filepath):
    """Read CSV line by line (memory efficient)"""
    with open(filepath) as f:
        header = f.readline()
        for line in f:
            yield line.strip().split(',')

# Usage
for row in read_large_csv('large_file.csv'):
    process_row(row)  # Only one row in memory at a time

# vs. List (memory intensive)
rows = [line.strip().split(',') for line in open('large_file.csv')]
for row in rows:
    process_row(row)  # Entire file in memory!
```

## Design Patterns for Data Engineering

### 1. Pipeline Pattern

```python
from abc import ABC, abstractmethod
from typing import Iterator, Any

class DataPipeline(ABC):
    """Abstract pipeline"""
    
    @abstractmethod
    def extract(self) -> Iterator[dict]:
        """Extract data from source"""
        pass
    
    @abstractmethod
    def transform(self, record: dict) -> dict:
        """Transform single record"""
        pass
    
    @abstractmethod
    def load(self, record: dict) -> None:
        """Load record to destination"""
        pass
    
    def run(self) -> None:
        """Execute pipeline"""
        for record in self.extract():
            try:
                transformed = self.transform(record)
                self.load(transformed)
            except Exception as e:
                self.handle_error(record, e)
    
    def handle_error(self, record: dict, error: Exception) -> None:
        """Error handling (can be overridden)"""
        raise error

class CSVToDataframePipeline(DataPipeline):
    """Concrete implementation"""
    
    def __init__(self, input_file: str, output_table: str):
        self.input_file = input_file
        self.output_table = output_table
        self.records_processed = 0
    
    def extract(self) -> Iterator[dict]:
        import pandas as pd
        df = pd.read_csv(self.input_file)
        for _, row in df.iterrows():
            yield row.to_dict()
    
    def transform(self, record: dict) -> dict:
        # Apply transformations
        record['amount'] = float(record['amount']) * 1.1
        record['processed_at'] = datetime.now().isoformat()
        return record
    
    def load(self, record: dict) -> None:
        # Write to database
        insert_to_database(record, self.output_table)
        self.records_processed += 1
    
    def handle_error(self, record: dict, error: Exception) -> None:
        # Log failed records to dead-letter queue
        log_failed_record(record, error)

# Usage
pipeline = CSVToDataframePipeline('input.csv', 'analytics.processed_data')
pipeline.run()
print(f"Processed: {pipeline.records_processed} records")
```

### 2. Strategy Pattern (for different data sources)

```python
from abc import ABC, abstractmethod

class DataExtractor(ABC):
    """Different extraction strategies"""
    
    @abstractmethod
    def extract(self) -> Iterator[dict]:
        pass

class CSVExtractor(DataExtractor):
    def __init__(self, filepath: str):
        self.filepath = filepath
    
    def extract(self) -> Iterator[dict]:
        import csv
        with open(self.filepath) as f:
            reader = csv.DictReader(f)
            for row in reader:
                yield row

class DatabaseExtractor(DataExtractor):
    def __init__(self, connection_string: str, query: str):
        self.connection_string = connection_string
        self.query = query
    
    def extract(self) -> Iterator[dict]:
        import psycopg2
        conn = psycopg2.connect(self.connection_string)
        cursor = conn.cursor()
        cursor.execute(self.query)
        columns = [desc[0] for desc in cursor.description]
        for row in cursor.fetchall():
            yield dict(zip(columns, row))
        cursor.close()
        conn.close()

class APIExtractor(DataExtractor):
    def __init__(self, api_url: str):
        self.api_url = api_url
    
    def extract(self) -> Iterator[dict]:
        import requests
        response = requests.get(self.api_url)
        for item in response.json():
            yield item

# Usage: Switch strategies without changing pipeline
extractors = {
    'csv': CSVExtractor('data.csv'),
    'database': DatabaseExtractor('postgres://...', 'SELECT * FROM users'),
    'api': APIExtractor('https://api.example.com/data')
}

def process_data(source_type: str):
    extractor = extractors[source_type]
    for record in extractor.extract():
        process_record(record)

process_data('csv')  # Use CSV extractor
process_data('database')  # Use DB extractor
```

### 3. Decorator Pattern (add functionality)

```python
from functools import wraps
from datetime import datetime
import time

def timing(func):
    """Decorator to time function execution"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        duration = time.time() - start
        print(f"{func.__name__} took {duration:.2f}s")
        return result
    return wrapper

def retry(max_attempts: int = 3, delay: int = 1):
    """Decorator to retry failed function"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except Exception as e:
                    if attempt == max_attempts - 1:
                        raise
                    print(f"Attempt {attempt + 1} failed, retrying in {delay}s...")
                    time.sleep(delay)
        return wrapper
    return decorator

def cache_result(func):
    """Decorator to cache function results"""
    cache = {}
    
    @wraps(func)
    def wrapper(*args, **kwargs):
        # Create cache key from arguments
        key = (args, tuple(kwargs.items()))
        
        if key not in cache:
            cache[key] = func(*args, **kwargs)
        
        return cache[key]
    
    return wrapper

# Usage
@timing
@retry(max_attempts=3, delay=2)
@cache_result
def expensive_operation(user_id: int) -> dict:
    """Load user data (with caching, retry, timing)"""
    # Could fail, but will retry
    return fetch_from_api(f'/users/{user_id}')

# Executes with all three decorators applied
user_data = expensive_operation(123)
```

## Performance Optimization

### 1. Vectorization (NumPy, Pandas)

```python
import pandas as pd
import numpy as np

# SLOW: Python loops
def calculate_discount_slow(prices):
    discounted = []
    for price in prices:
        if price > 100:
            discounted.append(price * 0.9)
        else:
            discounted.append(price)
    return discounted

# FAST: NumPy/Pandas vectorization
def calculate_discount_fast(prices):
    prices = pd.Series(prices)
    return np.where(prices > 100, prices * 0.9, prices)

# Performance comparison
prices = list(range(1_000_000))
%timeit calculate_discount_slow(prices)  # ~100 ms
%timeit calculate_discount_fast(prices)  # ~1 ms (100x faster!)

# Why? NumPy operations are implemented in C, not Python
# They operate on entire arrays at once, avoiding Python loops
```

### 2. Chunking for Large Datasets

```python
import pandas as pd

def process_large_csv(filepath: str, chunk_size: int = 10000):
    """Process large CSV in chunks (memory efficient)"""

        # Process each chunk
        chunk['amount'] = chunk['amount'] * 1.1
        chunk['processed_at'] = pd.Timestamp.now()

        load_to_database(chunk)

df = pd.read_csv('large_file.csv')  # If file > RAM, crashes!

```

## Distributed Computing with PySpark

### RDD vs. DataFrame

```python
from pyspark.sql import SparkSession
from pyspark.rdd import RDD

spark = SparkSession.builder.appName("demo").getOrCreate()

# RDD (Resilient Distributed Dataset): Low-level API
rdd = spark.sparkContext.parallelize([1, 2, 3, 4, 5])
result = rdd.map(lambda x: x * 2).filter(lambda x: x > 5).collect()
# Result: [6, 8, 10]

# DataFrame: High-level API (preferred)
data = [
    {'user_id': 1, 'amount': 100},
    {'user_id': 2, 'amount': 200},
    {'user_id': 1, 'amount': 50}
]

df = spark.createDataFrame(data)
result = (df
    .filter(df['amount'] > 75)
    .groupBy('user_id')
    .agg({'amount': 'sum'})
    .collect()
)

# DataFrame advantages:
# 1. Optimized by Catalyst optimizer
# 2. Lower memory usage (columnar storage)
# 3. SQL interface available
# 4. Type safety with schema
```

### Partitioning for Performance

```python
# Partition by key to enable efficient joins
orders = spark.read.parquet('orders/')
customers = spark.read.parquet('customers/')

# Without partitioning: Shuffle join (expensive)
result = orders.join(customers, 'customer_id')

# With partitioning: Much faster
orders_partitioned = orders.repartition('customer_id')
customers_partitioned = customers.repartition('customer_id')
result = orders_partitioned.join(customers_partitioned, 'customer_id')

# Best practice: Persist partitioned data
orders_partitioned.write.parquet(
    'orders_partitioned/',
    partitionBy='customer_id'  # Stored partitioned
)

# Later reads are automatically partitioned
orders = spark.read.parquet('orders_partitioned/')
```

### Broadcast Variables

```python
# Broadcast small reference data to all workers
reference_data = spark.read.parquet('lookup_table/').collect()
reference_dict = {row['id']: row['value'] for row in reference_data}

# Broadcast (send once to each worker)
broadcast_ref = spark.broadcast(reference_dict)

# Use in transformation
def enrich_with_reference(row):
    ref = broadcast_ref.value.get(row['ref_id'])
    return {**row, 'ref_value': ref}

enriched = df.rdd.map(enrich_with_reference).toDF()

# Benefits:
# - Without broadcast: Reference data shuffled for each partition
# - With broadcast: Each worker gets copy once, used many times
```

## Error Handling & Resilience

### Exception Handling

```python
class DataProcessingError(Exception):
    """Base exception for data processing"""
    pass

class ValidationError(DataProcessingError):
    """Data validation failed"""
    pass

class TransformationError(DataProcessingError):
    """Data transformation failed"""
    pass

def safe_transform(record: dict) -> dict:
    """Transform with comprehensive error handling"""
    
    try:
        # Validation
        if 'user_id' not in record:
            raise ValidationError("Missing user_id")
        
        if not isinstance(record['amount'], (int, float)):
            raise ValidationError(f"Invalid amount type: {type(record['amount'])}")
        
        # Transformation
        try:
            record['amount'] = float(record['amount']) * 1.1
        except (ValueError, TypeError) as e:
            raise TransformationError(f"Failed to transform amount: {e}")
        
        return record
    
    except ValidationError as e:
        logger.warning(f"Validation failed: {e}")
        # Send to quarantine queue
        send_to_quarantine(record, str(e))
        return None
    
    except TransformationError as e:
        logger.error(f"Transformation failed: {e}")
        # Send to dead-letter
        send_to_dlq(record, str(e))
        return None
    
    except Exception as e:
        logger.error(f"Unexpected error: {e}", exc_info=True)
        raise
```

### Idempotency

```python
# Idempotent operation: Can be retried safely

def idempotent_insert(record: dict, table_id: str):
    """Insert only if not already exists"""
    
    client = bigquery.Client()
    
    # Upsert (update if exists, insert if not)
    query = f"""
    MERGE INTO `{table_id}` T
    USING (
        SELECT '{record['id']}' as id, {record['amount']} as amount
    ) S
    ON T.id = S.id
    WHEN MATCHED THEN UPDATE SET amount = S.amount
    WHEN NOT MATCHED THEN INSERT (id, amount) VALUES (S.id, S.amount)
    """
    
    client.query(query).result()

# Idempotent version
# Can retry without creating duplicates
for attempt in range(3):
    try:
        idempotent_insert(record, 'my_project.analytics.data')
        break
    except Exception as e:
        if attempt == 2:
            raise
        time.sleep(2 ** attempt)  # Exponential backoff
```

## Interview Questions

### Q1: "Optimize Python code processing 1 billion rows from CSV."

**Answer**:
1. **Chunking**: Read in 100K chunks, not entire file
2. **Vectorization**: Use Pandas/NumPy instead of Python loops
3. **Multiprocessing**: Process chunks in parallel
4. **Lazy evaluation**: Generators for streaming
5. **Type hints**: Enable optimization, catch errors early

### Q2: "GIL limits multi-threading. How to parallelize?"

**Answer**:
- **I/O-bound** (network, disk): Use threading (GIL released during I/O)
- **CPU-bound** (math, processing): Use multiprocessing (separate GIL per process)
- **Hybrid**: Threading for I/O, multiprocessing for CPU

### Q3: "Design Python data pipeline for 100+ sources."

**Answer**:
1. **Strategy pattern**: Different extractors per source
2. **Pipeline pattern**: Reusable extract-transform-load
3. **Error handling**: Dead-letter queue for failures
4. **Observability**: Log metrics (processed count, duration)
5. **Idempotency**: Safe to retry
6. **Monitoring**: Alert on failures, track SLAs
