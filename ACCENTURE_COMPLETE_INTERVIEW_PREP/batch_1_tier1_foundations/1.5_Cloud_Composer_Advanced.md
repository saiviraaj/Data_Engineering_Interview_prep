## Advanced DAG Design Patterns

### 1. **Fan-Out / Fan-In Pattern**

Process multiple items in parallel, then aggregate results.

```python
from airflow import DAG
from airflow.operators.python import PythonOperator
from airflow.operators.dummy import DummyOperator
from airflow.models import Variable
from datetime import datetime, timedelta
import json

dag = DAG(
    'fan_out_fan_in_pattern',
    start_date=datetime(2024, 1, 1),
    schedule_interval='@daily',
    default_args={'retries': 2}
)

# Get list of items to process
def get_items_to_process(**context):
    """Generate list of items"""
    items = [
        {'id': 'source_1', 'table': 'raw_table_1'},
        {'id': 'source_2', 'table': 'raw_table_2'},
        {'id': 'source_3', 'table': 'raw_table_3'},
        # ... could be 100s of items
    ]
    context['task_instance'].xcom_push(
        key='items',
        value=json.dumps(items)
    )

setup = PythonOperator(
    task_id='setup',
    python_callable=get_items_to_process,
    provide_context=True,
    dag=dag
)

# FAN-OUT: Process each item
def process_item(item_id: str, **context):
    """Process single item"""
    print(f"Processing {item_id}")
    # Simulate work
    result = {
        'item_id': item_id,
        'rows_processed': 1000,
        'status': 'success'
    }
    context['task_instance'].xcom_push(
        key=f'result_{item_id}',
        value=json.dumps(result)
    )

# Create tasks dynamically
process_tasks = []
for i in range(1, 4):  # 3 sources
    task = PythonOperator(
        task_id=f'process_source_{i}',
        python_callable=process_item,
        op_kwargs={'item_id': f'source_{i}'},
        provide_context=True,
        dag=dag,
        pool_slots=2  # Limit concurrent execution
    )
    process_tasks.append(task)

# FAN-IN: Aggregate results
def aggregate_results(**context):
    """Combine results from all items"""
    ti = context['task_instance']
    
    all_results = []
    for i in range(1, 4):
        result = ti.xcom_pull(
            task_ids=f'process_source_{i}',
            key=f'result_source_{i}'
        )
        all_results.append(json.loads(result))
    
    total_rows = sum(r['rows_processed'] for r in all_results)
    print(f"Total rows processed: {total_rows}")
    
    ti.xcom_push(
        key='aggregated_result',
        value=json.dumps({'total': total_rows})
    )

aggregate = PythonOperator(
    task_id='aggregate',
    python_callable=aggregate_results,
    provide_context=True,
    dag=dag
)

finish = DummyOperator(
    task_id='finish',
    dag=dag
)

# Define dependencies
setup >> process_tasks >> aggregate >> finish

# Execution:
# setup (serial) → [process_source_1, process_source_2, process_source_3] (parallel) → aggregate → finish
```

### 3. **Sensor Pattern (Wait for Conditions)**

Wait for external events before proceeding.

```python
from airflow import DAG
from airflow.sensors.gcs import GCSObjectExistenceSensor
from airflow.sensors.sql import SqlCheckOperator
from airflow.operators.python import PythonOperator
from datetime import datetime, timedelta

dag = DAG(
    'sensor_pattern',
    start_date=datetime(2024, 1, 1),
    schedule_interval='@daily'
)

# Wait for file to appear in GCS
wait_for_file = GCSObjectExistenceSensor(
    task_id='wait_for_input_file',
    bucket='my-bucket',
    object='data/input_{{ execution_date.strftime("%Y%m%d") }}.csv',
    poke_interval=60,  # Check every 60 seconds
    timeout=3600,  # Fail if not found after 1 hour
    mode='reschedule',  # Free up worker while waiting
    dag=dag
)

# Wait for table to be populated
wait_for_data = SqlCheckOperator(
    task_id='wait_for_table_data',
    sql='''
        SELECT COUNT(*) FROM my_dataset.my_table
        WHERE DATE(created_at) = '{{ execution_date.strftime("%Y-%m-%d") }}'
    ''',
    conn_id='bigquery_default',
    hook_params={'use_legacy_sql': False},
    poke_interval=300,  # Check every 5 minutes
    dag=dag
)

# Process once conditions met
def process_data(**context):
    print("Conditions met! Processing data...")

process = PythonOperator(
    task_id='process',
    python_callable=process_data,
    dag=dag
)

wait_for_file >> wait_for_data >> process

# Sensor modes:
# - 'poke': Occupies worker while waiting (blocks)
# - 'reschedule': Releases worker, re-queues task later
# For long waits, use 'reschedule' to save resources
```

### 4. **Dag-to-Dag Dependencies**

Trigger one DAG from another.

```python

    task_id='trigger_child_dag',

task1 >> trigger_child >> task2

    'child_dag',
    start_date=datetime(2024, 1, 1),
    schedule_interval=None  # Only triggered by parent
)
```

## Production-Grade Error Handling

### 1. **Task-Level Retry Logic**

```python
from airflow import DAG
from airflow.operators.python import PythonOperator
from airflow.exceptions import AirflowException
from tenacity import (
    retry,
    wait_exponential,
    stop_after_attempt,
    retry_if_exception_type
)
from datetime import timedelta
import time

dag = DAG(
    'error_handling_dag',
    start_date=datetime(2024, 1, 1),
    default_args={
        'retries': 3,
        'retry_delay': timedelta(minutes=5)
    }
)

# Option 1: Airflow retry mechanism
@retry(
    wait=wait_exponential(multiplier=1, min=2, max=10),
    stop=stop_after_attempt(4),
    retry=retry_if_exception_type(ConnectionError)
)
def flaky_api_call():
    """API call that might fail"""
    import random
    if random.random() < 0.7:
        raise ConnectionError("Connection timeout")
    return "Success"

# Option 2: Custom retry logic
def robust_task_with_custom_retry(**context):
    """Task with custom retry logic"""
    max_attempts = 3
    attempt = 0
    
    while attempt < max_attempts:
        try:
            attempt += 1
            result = flaky_api_call()
            return result
        except ConnectionError as e:
            if attempt == max_attempts:
                raise AirflowException(f"Failed after {max_attempts} attempts: {e}")
            else:
                wait_time = 2 ** attempt  # Exponential backoff
                print(f"Attempt {attempt} failed, retrying in {wait_time}s...")
                time.sleep(wait_time)

task = PythonOperator(
    task_id='robust_task',
    python_callable=robust_task_with_custom_retry,
    dag=dag,
    retries=3,
    retry_delay=timedelta(minutes=2),
    # Backoff: 2^1=2s, 2^2=4s, 2^3=8s
)
```

### 2. **DAG-Level Callbacks**

```python
from airflow import DAG
from airflow.operators.dummy import DummyOperator

def on_failure_callback(context):

dag = DAG(
    'callback_dag',
```

### 3. **Exception Handling in PythonOperator**

```python
from airflow.exceptions import AirflowException
import logging

logger = logging.getLogger(__name__)

def error_aware_task(**context):
    """Task with comprehensive error handling"""
    ti = context['task_instance']
    
    try:
        # Main logic
        result = perform_complex_operation()
        
        # Push success result
        ti.xcom_push(key='status', value='success')
        ti.xcom_push(key='result', value=result)
        
    except ValueError as e:
        # Expected error - handle gracefully
        logger.warning(f"Data validation error: {e}")
        ti.xcom_push(key='status', value='warning')
        ti.xcom_push(key='error', value=str(e))
        
    except ConnectionError as e:
        # Transient error - let Airflow retry
        logger.error(f"Connection failed: {e}")
        raise AirflowException(f"Transient failure: {e}") from e
        
    except Exception as e:
        # Unexpected error
        logger.error(f"Unexpected error: {e}", exc_info=True)
        ti.xcom_push(key='status', value='error')
        raise AirflowException(f"Task failed: {e}") from e

def perform_complex_operation():
    """Actual work"""
    import random
    if random.random() < 0.3:
        raise ConnectionError("API unreachable")
    return "Data processed"
```

## Interview Scenarios

### Q1: "Design a DAG for real-time event processing with 1 million events/second."

**Answer**:
- **Not batch-based**: Cloud Composer (Airflow) is for batch/scheduled workflows
- **Use streaming architecture**:
  - Cloud Pub/Sub for ingestion (1M events/sec)
  - Dataflow for processing (auto-scaling)
  - BigQuery for storage (streaming inserts)
  - Cloud Composer for orchestration (schedule periodic aggregation)

### Q2: "How to handle late-arriving data in Airflow?"

**Answer**:
1. **Watermark approach**: Track data freshness
2. **Flexible partitioning**: Create new partition if data late
3. **Idempotent transforms**: Handle duplicate processing
4. **Backfill capability**: Re-process partition if needed

### Q3: "DAG runs 1 hour but SLA is 30 minutes. How to optimize?"

**Answer**:
1. **Identify bottleneck**: Check execution plan
2. **Parallelize tasks**: Run independent tasks together
3. **Optimize individual tasks**: Query optimization, resource allocation
4. **Reduce pool contention**: Increase pool slots for critical tasks
5. **Cache/materialize**: Pre-compute expensive operations
